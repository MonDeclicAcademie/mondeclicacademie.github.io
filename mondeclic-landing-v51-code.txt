/* ---------- the light ---------- */
(function () {
  const root = document.documentElement;
  if (window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches) return; // static version
  root.classList.add("reveal");

  const $ = function (s) { return document.querySelector(s); };
  const lamp = $(".lamp"), head = $("#lhead"), bulbOn = $("#bulbOn"), rimGlow = $("#rimGlow");
  const cone = $(".cone"), graze = $(".graze"), core = $(".cone-core"), halo = $(".halo");
  const dimU = $(".dim-u"), dimR = $(".dim-r"), wash = $(".wash"), prog = $(".prog");
  const cue = $("#cue"), heroSpark = $("#heroSpark");
  const hlogo = $(".hlogo"), bmOn = $("#bmOn"), bGlow = $("#bGlow");
  const rail = $("#rail"), steps = Array.prototype.slice.call(document.querySelectorAll(".step"));
  const dust = $("#dust"), ctx = dust.getContext("2d");

  let veiled = Array.prototype.slice.call(document.querySelectorAll("[data-l]"));
  veiled.forEach(function (el) {
    const sibs = Array.prototype.filter.call(el.parentElement.children, function (c) { return c.hasAttribute("data-l"); });
    const i = sibs.indexOf(el);
    if (i > 0) el.style.transitionDelay = Math.min(i, 4) * 110 + "ms";
  });

  const sm = function (t) { t = t < 0 ? 0 : t > 1 ? 1 : t; return t * t * (3 - 2 * t); };
  let vw, vh, docH, lr, k, dpr;
  const L = { on: 0, ax: 0, ay: 0, th: 0 };
  const tracks = Array.prototype.map.call(document.querySelectorAll("[data-track]"), function (el) {
    return { el: el, items: el.querySelectorAll(".ti"), keep: el.hasAttribute("data-keep") };
  });

  function measure() {
    vw = window.innerWidth; vh = window.innerHeight;
    docH = document.documentElement.scrollHeight;
    lr = lamp.getBoundingClientRect(); k = lr.width / 260;
    dpr = Math.min(1.5, window.devicePixelRatio || 1);
    dust.width = Math.round(vw * dpr); dust.height = Math.round(vh * dpr);
  }

  let queued = false;
  function frame() {
    queued = false;
    const y = window.scrollY || window.pageYOffset;
    const q = Math.min(1, Math.max(0, y / Math.max(1, docH - vh)));
    const on = sm((y - vh * 0.04) / (vh * 0.45));

    const deg = 28 + 8 * q, th = deg * Math.PI / 180;
    head.setAttribute("transform", "translate(96 60) rotate(" + deg.toFixed(2) + ")");
    const ax = lr.left + (96 - 66 * Math.sin(th)) * k;
    const ay = lr.top + (60 + 66 * Math.cos(th)) * k;
    L.on = on; L.ax = ax; L.ay = ay; L.th = th;

    const py = vh * 0.55;
    let px = ax - ((py - ay) / Math.cos(th)) * Math.sin(th);
    px = Math.max(vw * 0.3, Math.min(px, vw * 0.6));

    const dim = 0.2 * (1 - sm(q / 0.9));
    dimU.style.opacity = (dim * (1 - on)).toFixed(3);
    dimR.style.opacity = (dim * on).toFixed(3);
    dimR.style.transform = "translate(" + px.toFixed(1) + "px," + py.toFixed(1) + "px) scale(" + (1 + 1.8 * q).toFixed(3) + ")";

    const t = "translate(" + ax.toFixed(1) + "px," + ay.toFixed(1) + "px) rotate(" + deg.toFixed(2) + "deg)";
    const spread = " scale(" + (0.9 + 0.45 * q).toFixed(3) + ")";
    cone.style.transform = t + spread; cone.style.opacity = (on * 0.95).toFixed(3);
    graze.style.transform = t + spread; graze.style.opacity = (on * 0.75).toFixed(3);
    core.style.transform = t; core.style.opacity = on.toFixed(3);
    halo.style.transform = "translate(" + ax.toFixed(1) + "px," + ay.toFixed(1) + "px)"; halo.style.opacity = on.toFixed(3);

    bulbOn.setAttribute("opacity", on.toFixed(3));
    rimGlow.setAttribute("opacity", (on * 0.75).toFixed(3));
    wash.style.opacity = sm((q - 0.72) / 0.28).toFixed(3);
    prog.style.transform = "scaleX(" + q.toFixed(4) + ")";
    cue.style.opacity = (1 - sm(on * 1.6)).toFixed(3);
    heroSpark.classList.toggle("on", on > 0.55);

    // the déclic: the very first scroll switches the logo on, while it is still in view
    // the small logo in the header follows the lamp, and stays visible all along
    bmOn.style.opacity = on.toFixed(3);
    bGlow.style.opacity = on.toFixed(3);

    // the line of light travels through the three steps, lighting each one as it arrives
    const rr = rail.getBoundingClientRect();
    const f = sm((vh * 0.78 - rr.top) / (rr.height + vh * 0.28));
    rail.style.setProperty("--f", f.toFixed(3));
    for (let i = 0; i < steps.length; i++) steps[i].classList.toggle("on", f >= (i + 0.35) / steps.length);

    // reading tracks: the light rests on the line nearest the middle of the screen
    for (let t = 0; t < tracks.length; t++) {
      const tr = tracks[t], r = tr.el.getBoundingClientRect(), fl = vh * 0.5;
      const inside = r.top < fl && r.bottom > fl - 60;
      let best = -1, bd = 1e9;
      for (let i = 0; i < tr.items.length; i++) {
        const ir = tr.items[i].getBoundingClientRect();
        const d = Math.abs(ir.top + ir.height / 2 - fl);
        if (d < bd) { bd = d; best = i; }
        if (tr.keep && ir.top < fl) tr.items[i].classList.add("on");
      }
      for (let i = 0; i < tr.items.length; i++) {
        const f = inside && i === best;
        tr.items[i].classList.toggle("focus", f);
        if (f) tr.items[i].classList.add("seen");
      }
      if (tr.keep) tr.el.style.setProperty("--t", sm((fl - r.top) / Math.max(1, r.height)).toFixed(3));
    }

    if (veiled.length) {
      const line = vh * 0.8;
      veiled = veiled.filter(function (el) {
        if (el.getBoundingClientRect().top < line) { el.classList.add("lit"); return false; }
        return true;
      });
    }
    if (on > 0.02) startDust();
  }
  function request() { if (!queued) { queued = true; requestAnimationFrame(frame); } }

  /* ---- dust drifting in the beam ---- */
  const motes = [];
  function mote(fresh) {
    return { d: fresh ? 0.03 : Math.random(), s: Math.random() * 2 - 1, r: 0.5 + Math.random() * 1.3,
             v: 0.00006 + Math.random() * 0.00012, w: Math.random() * 6.28, a: 0.3 + Math.random() * 0.5 };
  }
  for (let i = 0; i < 46; i++) motes.push(mote(false));
  let dustOn = false, last = 0;
  function startDust() { if (!dustOn && !document.hidden) { dustOn = true; last = performance.now(); requestAnimationFrame(drawDust); } }
  function drawDust(now) {
    const dt = Math.min(48, now - last); last = now;
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    ctx.clearRect(0, 0, vw, vh);
    if (L.on <= 0.02 || document.hidden) { dustOn = false; return; }
    const len = Math.max(vw, vh) * 0.95, sn = Math.sin(L.th), cs = Math.cos(L.th), spread = Math.tan(24 * Math.PI / 180);
    ctx.fillStyle = "#FFF3CF";
    for (let i = 0; i < motes.length; i++) {
      const m = motes[i];
      m.d += m.v * dt; m.w += 0.0009 * dt; m.s += Math.sin(m.w) * 0.0006 * dt * 0.06;
      if (m.d > 1 || Math.abs(m.s) > 1) { motes[i] = mote(true); continue; }
      const dist = m.d * len, lat = m.s * dist * spread;
      const x = L.ax - sn * dist + cs * lat, y = L.ay + cs * dist + sn * lat;
      const fade = Math.min(1, m.d / 0.08) * Math.pow(1 - m.d, 1.3) * (1 - m.s * m.s);
      ctx.globalAlpha = m.a * fade * L.on;
      ctx.beginPath(); ctx.arc(x, y, m.r, 0, 6.2832); ctx.fill();
    }
    ctx.globalAlpha = 1;
    requestAnimationFrame(drawDust);
  }
  document.addEventListener("visibilitychange", function () { if (!document.hidden && L.on > 0.02) startDust(); });

  document.addEventListener("focusin", function (e) {
    const el = e.target.closest && e.target.closest("[data-l]");
    if (el && !el.classList.contains("lit")) { el.classList.add("lit"); veiled = veiled.filter(function (v) { return v !== el; }); }
  });

  measure(); frame();
  window.addEventListener("scroll", request, { passive: true });
  window.addEventListener("resize", function () { measure(); request(); });
  window.addEventListener("load", function () { measure(); request(); });
})();

  const root = document.documentElement;

  const num = (CONFIG.whatsapp || "").replace(/\D/g, "");
  const KEY = "mondeclic-formulaire-v1";
  let cur = 1;

  /* open / close */
  function openForm() {
    if (!dlg.open) { try { dlg.showModal(); } catch (e) { dlg.setAttribute("open", ""); } }
    root.classList.add("modal-open");
    show(cur, false);
    setTimeout(function () { const h = document.getElementById("fTitle"); h.setAttribute("tabindex", "-1"); h.focus({ preventScroll: true }); }, 30);
  }
  function closeForm() { if (typeof dlg.close === "function" && dlg.open) dlg.close(); else dlg.removeAttribute("open"); root.classList.remove("modal-open"); }
  document.querySelectorAll("[data-form]").forEach(function (b) {
    b.addEventListener("click", function (e) { e.preventDefault(); openForm(); });
  });
  document.getElementById("fClose").addEventListener("click", closeForm);
  dlg.addEventListener("close", function () { root.classList.remove("modal-open"); });
  dlg.addEventListener("click", function (e) { if (e.target === dlg) closeForm(); });

  /* answers, kept as a draft on this device only */
  function read() {
    const d = {};
    Array.prototype.forEach.call(form.elements, function (el) {
      if (!el.name) return;
      if (el.type === "checkbox") { (d[el.name] = d[el.name] || []); if (el.checked) d[el.name].push(el.value); }
      else if (el.type === "radio") { if (el.checked) d[el.name] = el.value; }
      else d[el.name] = el.value;
    });
    return d;
  }
  function restore(d) {
    Array.prototype.forEach.call(form.elements, function (el) {
      if (!el.name || !(el.name in d)) return;
      if (el.type === "checkbox") el.checked = (d[el.name] || []).indexOf(el.value) > -1;
      else if (el.type === "radio") el.checked = d[el.name] === el.value;
      else el.value = d[el.name];
    });
  }
  let saveTimer;
  function save() {
    clearTimeout(saveTimer);
    saveTimer = setTimeout(function () {
      try { localStorage.setItem(KEY, JSON.stringify({ d: read(), s: cur })); saved.textContent = "Brouillon enregistré sur cet appareil"; } catch (e) {}
    }, 400);
  }

  /* one comfort scale per chosen subject */
  function buildScales(keep) {
    const subjects = read().matieres || [];
    const list = subjects.length ? subjects : ["La matière concernée"];
